% Flexible two-column row. Argument 1 is the left fraction AFTER the gutter.
% Both columns use natural height; the next row follows the taller column.
% Example: \PosterColumns{0.5}{left content}{right content}
\newlength{\PosterRowWidth}
\newlength{\PosterLeftWidth}
\newlength{\PosterRightWidth}
\newcommand{\PosterColumns}[3]{%
  \par\begingroup
  \setlength{\PosterRowWidth}{\dimexpr\linewidth-\PosterGutter\relax}%
  \setlength{\PosterLeftWidth}{#1\PosterRowWidth}%
  \setlength{\PosterRightWidth}{\dimexpr\PosterRowWidth-\PosterLeftWidth\relax}%
  \noindent
  \begin{minipage}[t]{\PosterLeftWidth}\vspace{0pt}\raggedright #2\par\end{minipage}%
  \hspace{\PosterGutter}%
  \begin{minipage}[t]{\PosterRightWidth}\vspace{0pt}\raggedright #3\par\end{minipage}%
  \par\endgroup}

% Card height at A1, in millimetres.
\newcommand{\PosterResultHeightMM}{120}

% Arguments: colour, parameter, classification, description, image.
\newcommand{\PosterResult}[5]{%
  \begin{tcolorbox}[
    colback=#1!7!white,
    colframe=#1!7!white,
    boxrule=0pt,
    arc=1.5\PosterUnit,
    left=7\PosterUnit,
    right=7\PosterUnit,
    top=6\PosterUnit,
    bottom=6\PosterUnit,
    boxsep=0pt,
    before skip=0pt,
    after skip=0pt,
    height=\PosterResultHeightMM\PosterUnit,
    valign=center
  ]
    \noindent
    % Everything textual goes in the left column.
    \begin{minipage}[c]{0.34\linewidth}
      \raggedright
      {\PosterLabelFont\color{#1}#2\par}
      \PosterSpace{3}
      {\PosterFont{39}{46}\bfseries\color{#1}#3\par}
      \PosterSpace{3}
      {\PosterFont{21}{28}#4\par}
    \end{minipage}%
    \hfill
    % The image can use almost the entire card height.
    \begin{minipage}[c]{0.62\linewidth}
      \centering
      \includegraphics[
        width=\linewidth,
        height=\dimexpr
          \PosterResultHeightMM\PosterUnit-16\PosterUnit
        \relax,
        keepaspectratio
      ]{#5}%
    \end{minipage}%
  \end{tcolorbox}%
}

% Loads a real figure if the file exists, otherwise reserves the same height.
% Arguments: filename, A1 height in mm (number only), placeholder label.
\newcommand{\PosterFigure}[3]{%
  \begin{tcolorbox}[colback=PosterFigure,colframe=PosterLine,
    boxrule=0.45\PosterPoint,arc=1.5\PosterUnit,
    left=4\PosterUnit,right=4\PosterUnit,
    top=4\PosterUnit,bottom=4\PosterUnit,boxsep=0pt,
    before skip=0pt,after skip=0pt,height=#2\PosterUnit,
    valign=center,halign=center]
    \IfFileExists{#1}{%
      \includegraphics[width=\linewidth,
        height=\dimexpr#2\PosterUnit-9\PosterUnit\relax,
        keepaspectratio]{#1}%
    }{%
      \ifPosterShowFigurePlaceholders
        {\PosterLabelFont\color{PosterMuted}FIGURE SPACE\par}\PosterSpace{3}
        {\PosterSmallFont\color{PosterMuted}#3\par}%
      \else\mbox{}\fi
    }
  \end{tcolorbox}}
